        item {
            SectionCard(title = "公式画布") {
                Text(
                    text = "从左侧素材栏拖入节点，点击节点进行配置。",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(500.dp),
                    horizontalArrangement = Arrangement.spacedBy(0.dp),
                ) {
                    RuleCanvasPalette(
                        onPaletteItemSelected = { item ->
                            when (item) {
                                is PaletteItem.EmptySlot -> {
                                    val direction = if (item.label.contains("输出")) {
                                        RuleSlotDirection.OUTPUT
                                    } else {
                                        RuleSlotDirection.INPUT
                                    }
                                    val sourceType = when {
                                        item.label.contains("系统") -> RuleSlotSourceType.SYSTEM_INPUT
                                        item.label.contains("固定") -> RuleSlotSourceType.CONFIG_INPUT
                                        direction == RuleSlotDirection.OUTPUT -> RuleSlotSourceType.ATTRIBUTE_OUTPUT
                                        else -> RuleSlotSourceType.ATTRIBUTE_INPUT
                                    }
                                    addSlot(direction, sourceType)
                                }
                                is PaletteItem.Operator -> {
                                    val newNode = CanvasNode(
                                        type = CanvasNodeType.OPERATOR,
                                        position = Offset(100f + canvasState.nodes.size * 150f, 100f),
                                        operatorSymbol = item.symbol,
                                    )
                                    canvasState = canvasState.copy(nodes = canvasState.nodes + newNode)
                                }
                                is PaletteItem.Function -> {
                                    val newNode = CanvasNode(
                                        type = CanvasNodeType.FUNCTION,
                                        position = Offset(100f + canvasState.nodes.size * 150f, 100f),
                                        functionName = item.name,
                                    )
                                    canvasState = canvasState.copy(nodes = canvasState.nodes + newNode)
                                }
                            }
                        },
                    )
                    DraggableRuleCanvas(
                        canvasState = canvasState,
                        onNodeAdded = { node ->
                            canvasState = canvasState.copy(nodes = canvasState.nodes + node)
                        },
                        onNodeMoved = { nodeId, newPosition ->
                            canvasState = canvasState.copy(
                                nodes = canvasState.nodes.map { node ->
                                    if (node.id == nodeId) node.copy(position = newPosition) else node
                                }
                            )
                        },
                        onNodeSelected = { nodeId ->
                            canvasState = canvasState.copy(selectedNodeId = nodeId)
                            val node = canvasState.nodes.find { it.id == nodeId }
                            if (node?.type == CanvasNodeType.SLOT && node.slotConfig != null) {
                                showSlotConfigDialog = nodeId
                            }
                        },
                        onNodeDeleted = { nodeId ->
                            removeSlot(nodeId)
                        },
                        onConnectionAdded = { fromId, toId ->
                            val newConnection = CanvasConnection(fromNodeId = fromId, toNodeId = toId)
                            canvasState = canvasState.copy(connections = canvasState.connections + newConnection)
                        },
                        modifier = Modifier.weight(1f),
                    )
                }

                Spacer(modifier = Modifier.height(12.dp))
                Text(
                    text = "公式编码（高级）",
                    style = MaterialTheme.typography.labelLarge,
                    fontWeight = FontWeight.SemiBold,
                )

                    value = expressionField,
                    onValueChange = { expressionField = it },

                    },
                    minLines = 3,
                )
                RuleWorkbenchDecoderCard(decoder = decoder)
            }
